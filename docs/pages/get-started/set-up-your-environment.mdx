// - Menu edit (add / update / remove)

const STORAGE_KEYS = {
  menu: 'af_menu_v1',
  orders: 'af_orders_v1',
  sales: 'af_sales_v1',
  inventoryEnabled: 'af_inv_enabled_v1'
};

// --- default menu (from your message) ---
const DEFAULT_MENU = [
  { id: idgen(), name: 'চাউমিন (Chowmein)', priceMin:30, priceMax:50, price:40, stock: null },
  { id: idgen(), name: 'রুটি প্লেট (Ruti Plate)', price:50, priceMin:null, priceMax:null, stock: null },
  { id: idgen(), name: 'পুরি প্লেট (Puri Plate)', price:50, priceMin:null, priceMax:null, stock: null },
  { id: idgen(), name: 'সমোসা (Samusa)', price:10, priceMin:null, priceMax:null, stock: null },
  { id: idgen(), name: 'পকোড়া প্লেট (Pokora Plate)', price:30, priceMin:null, priceMax:null, stock: null },
  { id: idgen(), name: 'চিকেন মোমোস প্লেট (Chicken Momos Plate)', price:50, priceMin:null, priceMax:null, stock: null }
];

function idgen(){ return 'id_' + Math.random().toString(36).slice(2,9); }

// --- storage helpers ---
function save(key, val){ localStorage.setItem(key, JSON.stringify(val)); }
function load(key, fallback){ const raw = localStorage.getItem(key); return raw ? JSON.parse(raw) : fallback; }

// --- initialize data ---
let menu = load(STORAGE_KEYS.menu, DEFAULT_MENU);
let orders = load(STORAGE_KEYS.orders, []);
let sales = load(STORAGE_KEYS.sales, []); // each sale {id, dateISO, itemId, qty, total, name}
let inventoryEnabled = load(STORAGE_KEYS.inventoryEnabled, false);

function showView(name){
  views.forEach(v=>{
    v.classList.toggle('hidden', !v.id.endsWith(name));
  });
  // refresh content when switching
  renderMenu();
  renderOrders();
  renderSales();
  renderInventory();
}

// --- menu functions ---
function renderMenu(){
  // populate select for ordering
  orderItemSelect.innerHTML = '';
  menu.forEach(it=>{
    const opt = document.createElement('option');
    opt.value = it.id;
    opt.textContent = `${it.name} — ₹${it.price}${it.priceMin ? ` (₹${it.priceMin}-₹${it.priceMax||it.price})` : ''}`;
    orderItemSelect.appendChild(opt);
  });

  // render menu list with edit/delete
  menuList.innerHTML = '';
  menu.forEach(it=>{
    const div = document.createElement('div');
    div.className = 'item';
    const left = document.createElement('div'); left.className='meta';
    left.innerHTML = `<div><strong>${escapeHtml(it.name)}</strong><div class="small">₹${it.price}${it.stock !== null ? ' — stock: '+it.stock : ''}</div></div>`;
    const right = document.createElement('div');
    const edit = document.createElement('button'); edit.textContent='Edit';
    const del = document.createElement('button'); del.textContent='Delete';
    edit.addEventListener('click', ()=> {
      menuName.value = it.name; menuPrice.value = it.price; menuStock.value = (it.stock===null?'':it.stock);
      menuForm.dataset.editId = it.id;
      menuName.focus();
    });
    del.addEventListener('click', ()=> {
      if(confirm(`"${it.name}" মুছে ফেলতে চান?`)){
        menu = menu.filter(x=>x.id !== it.id);
        save(STORAGE_KEYS.menu, menu);
        renderMenu();
      }
    });
    right.appendChild(edit); right.appendChild(del);
    div.appendChild(left); div.appendChild(right);
    menuList.appendChild(div);
  });
}

menuForm.addEventListener('submit', e=>{
  e.preventDefault();
  const name = menuName.value.trim();
  const price = Number(menuPrice.value);
  const stockVal = menuStock.value.trim();
  const stock = stockVal === '' ? null : Number(stockVal);
  if(!name || !price) return alert('সঠিক নাম ও দাম দিন।');

  if(menuForm.dataset.editId){
    // update
    const id = menuForm.dataset.editId;
    const it = menu.find(x=>x.id===id);
    if(it){
      it.name = name; it.price = price; it.stock = stock;
    }
    delete menuForm.dataset.editId;
  } else {
    // add new
    menu.push({ id: idgen(), name, price, stock });
  }
  save(STORAGE_KEYS.menu, menu);
  menuForm.reset();
  renderMenu();
  renderOrders();
});

resetMenuBtn.addEventListener('click', ()=> menuForm.reset());

// --- orders functions ---
orderForm.addEventListener('submit', e=>{
  e.preventDefault();
  const cust = document.getElementById('cust-name').value.trim();
  const itemId = orderItemSelect.value;
  const qty = Number(document.getElementById('order-qty').value);
  const note = document.getElementById('order-note').value.trim();

  const item = menu.find(m=>m.id===itemId);
  if(!item) return alert('আইটেম পাওয়া যায়নি।');

  if(inventoryEnabled && item.stock !== null){
    if(item.stock < qty) return alert('স্টক পর্যাপ্ত নেই।');
    item.stock -= qty;
    save(STORAGE_KEYS.menu, menu);
  }

  const order = {
    id: idgen(),
    customer: cust || 'গেস্ট',
    itemId,
    itemName: item.name,
    qty,
    note,
    status: 'preparing', // or ready
    createdAt: new Date().toISOString()
  };
  orders.unshift(order);
  save(STORAGE_KEYS.orders, orders);

  // record sale immediately (you may choose to record on status ready; here we record now)
  const sale = {
    id: idgen(),
    dateISO: new Date().toISOString(),
    itemId: item.id,
    name: item.name,
    qty,
    total: item.price * qty
  };
  sales.push(sale);
  save(STORAGE_KEYS.sales, sales);

  orderForm.reset();
  renderOrders();
  renderSales();
  renderMenu();
});

bulkSample.addEventListener('click', ()=>{
  // add a demo order for convenience
  if(menu.length === 0) return alert('মেনু খালি।');
  document.getElementById('cust-name').value = 'Demo Customer';
  document.getElementById('order-qty').value = 1;
  orderItemSelect.selectedIndex = 0;
});

// render orders list
function renderOrders(){
  ordersList.innerHTML = '';
  if(orders.length === 0) {
    ordersList.innerHTML = '<div class="card small">কোন অর্ডার নেই</div>';
    return;
  }
  orders.forEach(o=>{
    const div = document.createElement('div'); div.className='item';
    const left = document.createElement('div'); left.innerHTML = `<div><strong>${escapeHtml(o.itemName)}</strong> <div class="small">পরিমাণ: ${o.qty} — ${escapeHtml(o.customer)}</div></div>`;
    const right = document.createElement('div');
    const statusBadge = document.createElement('span'); statusBadge.className='badge ' + (o.status==='preparing' ? 'status-prep' : 'status-ready');
    statusBadge.textContent = o.status === 'preparing' ? 'প্রস্তুত হচ্ছে' : 'প্রস্তুত';
    const toggle = document.createElement('button'); toggle.textContent = o.status === 'preparing' ? 'Mark ready' : 'Mark preparing';
    const remove = document.createElement('button'); remove.textContent = 'Remove';
    toggle.addEventListener('click', ()=> {
      o.status = o.status === 'preparing' ? 'ready' : 'preparing';
      save(STORAGE_KEYS.orders, orders);
      renderOrders();
      renderSales();
    });
    remove.addEventListener('click', ()=> {
      if(confirm('এই অর্ডার মুছতে চান?')){
        orders = orders.filter(x=>x.id !== o.id);
        save(STORAGE_KEYS.orders, orders);
        renderOrders();
      }
    });

    right.appendChild(statusBadge); right.appendChild(toggle); right.appendChild(remove);
    div.appendChild(left); div.appendChild(right);
    ordersList.appendChild(div);
  });
}

// --- sales functions ---
function renderSales(){
  // today's total
  const today = new Date().toISOString().slice(0,10);
  const todaySales = sales.filter(s => s.dateISO.slice(0,10) === today);
  const total = todaySales.reduce((acc,s)=>acc + s.total, 0);
  todaySalesEl.textContent = `₹${total}`;

  totalOrdersEl.textContent = sales.length;

  // top item overall
  const agg = {};
  sales.forEach(s => { agg[s.name] = (agg[s.name]||0) + s.qty; });
  const top = Object.keys(agg).sort((a,b)=>agg[b]-agg[a])[0] || '—';
  topItemEl.textContent = top;

  // sales list
  salesList.innerHTML = '';
  if(sales.length === 0) salesList.innerHTML = '<div class="card small">কোন বিক্রয় রেকর্ড নেই</div>';
  else {
    sales.slice().reverse().forEach(s=>{
      const div = document.createElement('div'); div.className='item';
      div.innerHTML = `<div><strong>${escapeHtml(s.name)}</strong><div class="small">${s.qty} pcs — ₹${s.total} — ${new Date(s.dateISO).toLocaleString()}</div></div><div><small>id:${s.id}</small></div>`;
      salesList.appendChild(div);
    });
  }
}

// --- inventory functions ---
inventoryToggle.checked = inventoryEnabled;
inventoryToggle.addEventListener('change', ()=>{
  inventoryEnabled = inventoryToggle.checked;
  save(STORAGE_KEYS.inventoryEnabled, inventoryEnabled);
  renderInventory();
});
function renderInventory(){
  inventoryList.innerHTML = '';
  if(!inventoryEnabled){
    inventoryList.innerHTML = '<div class="card small">ইনভেন্টরি নিষ্ক্রিয় আছে — চালু করতে চেকবক্সটি টিক করুন।</div>';
    return;
  }
  if(menu.length === 0) {
    inventoryList.innerHTML = '<div class="card small">মেনু খালি — আইটেম যোগ করুন এবং স্টক সেট করুন।</div>';
    return;
  }
  menu.forEach(it=>{
    const div = document.createElement('div'); div.className='item';
    const left = document.createElement('div'); left.innerHTML = `<div><strong>${escapeHtml(it.name)}</strong><div class="small">Stock: ${it.stock===null? 'Unset' : it.stock}</div></div>`;
    const right = document.createElement('div');
    const inc = document.createElement('button'); inc.textContent='+';
    const dec = document.createElement('button'); dec.textContent='-';
    const set = document.createElement('button'); set.textContent='Set';
    inc.addEventListener('click', ()=> {
      it.stock = (it.stock||0) + 1; save(STORAGE_KEYS.menu, menu); renderInventory(); renderMenu();
    });
    dec.addEventListener('click', ()=> {
      it.stock = Math.max(0, (it.stock||0) - 1); save(STORAGE_KEYS.menu, menu); renderInventory(); renderMenu();
    });
    set.addEventListener('click', ()=> {
      const v = prompt('নতুন স্টক সংখ্যা দিন (সংখ্যা):', it.stock===null?'0':it.stock);
      if(v !== null){
        const nv = Number(v);
        if(!Number.isFinite(nv) || nv < 0) return alert('সঠিক সংখ্যা দিন।');
        it.stock = nv; save(STORAGE_KEYS.menu, menu); renderInventory(); renderMenu();
      }
    });
    right.appendChild(inc); right.appendChild(dec); right.appendChild(set);
    div.appendChild(left); div.appendChild(right);
    inventoryList.appendChild(div);
  });
}

// --- utilities ---
function escapeHtml(s){ return (s+'').replace(/[&<>"']/g, c=>({ '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[c])); }

// initial render
renderMenu(); renderOrders(); renderSales(); renderInventory();
